# Delete SageMaker Endpoints, Configs, and Models
$ENDPOINTS = (aws sagemaker list-endpoints --region $REGION --query "Endpoints[*].EndpointName" --output text)
foreach ($ep in $ENDPOINTS -split '\s+') {
    if ($ep) { Write-Host "Deleting SageMaker Endpoint: $ep"; aws sagemaker delete-endpoint --endpoint-name $ep --region $REGION }
}

$CONFIGS = (aws sagemaker list-endpoint-configs --region $REGION --query "EndpointConfigs[*].EndpointConfigName" --output text)
foreach ($cfg in $CONFIGS -split '\s+') {
    if ($cfg) { aws sagemaker delete-endpoint-config --endpoint-config-name $cfg --region $REGION }
}

$MODELS = (aws sagemaker list-models --region $REGION --query "Models[*].ModelName" --output text)
foreach ($m in $MODELS -split '\s+') {
    if ($m) { aws sagemaker delete-model --model-name $m --region $REGION }
}

$LAMBDAS = (aws lambda list-functions --region $REGION --query "Functions[?contains(FunctionName, 'remediat') || contains(FunctionName, 'eks') || contains(FunctionName, 'security')].FunctionName" --output text)
foreach ($fn in $LAMBDAS -split '\s+') {
    if ($fn) { Write-Host "Deleting Lambda: $fn"; aws lambda delete-function --function-name $fn --region $REGION }
}
